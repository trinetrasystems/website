DeepStream Autostart

Jetson AGX Orin · systemd
deepstream
Setup — run once
cat > ~/start_deepstream.sh << 'EOF'
#!/bin/bash
exec deepstream-app \
  -c ~/DeepStream-Yolo/deepstream_app_config.txt \
  2>&1 | tee ~/deepstream.log
EOF
chmod +x ~/start_deepstream.sh

sudo tee /etc/systemd/system/deepstream.service << 'EOF'
[Unit]
Description=DeepStream YOLO Pipeline
After=network-online.target nvargus-daemon.service
Wants=network-online.target
Requires=nvargus-daemon.service

[Service]
Type=simple
User=nvidia
WorkingDirectory=/home/nvidia
ExecStart=/home/nvidia/start_deepstream.sh
Restart=on-failure
RestartSec=10
LimitNOFILE=65536

[Install]
WantedBy=multi-user.target
EOF

sudo systemctl daemon-reload
Commands
sudo systemctl start deepstream.service start
sudo systemctl stop deepstream.service stop
sudo systemctl restart deepstream.service restart
sudo systemctl enable deepstream.service enable
sudo systemctl disable deepstream.service disable
sudo systemctl status deepstream.service status
tail -f ~/deepstream.log | grep -i "fps\|error\|warn" logs
